Private Sub LimpiarTodoAccess_Click()
 ' Mostrar un mensaje de confirmación al usuario
  If MsgBox("¿Estás seguro de que quieres eliminar todos los datos de las tablas?", vbYesNo + vbCritical, "Confirmación") = vbYes Then

    ' Conectar a la base de datos
    conectarBD

    ' Eliminar los datos de las tablas
    Dim tablas As Variant
    tablas = Array("PESAJE_TALCAHUANO", "PESAJE_PLACILLA_2", "PESAJE_SANTIAGO", "PESAJE_PLACILLA")

    On Error Resume Next ' Ignorar errores si alguna tabla no existe
    For Each tabla In tablas
      CurrentDb.Execute "DELETE * FROM " & tabla & ";"
    Next tabla
    On Error GoTo 0 ' Restaurar el manejo de errores normal

    ' Desconectar de la base de datos
    desconectarBD

    ' Mostrar un mensaje de éxito al usuario
    MsgBox "Se eliminaron todos los datos de las tablas.", vbInformation, "Eliminación completa"

  End If

End Sub

Public Function CARGAR_INF_PRECIOS_TEO_PL(grilla As MSHFlexGrid, fecha_ini As Date, fecha_term As Date)
Dim rst As ADODB.Recordset
Set rst = New ADODB.Recordset
Dim row, fila As Integer
Dim sql, precio As String
Dim total_kilos, total_pesos As Double
Dim kilos_recam, kilos_super_basulip, kilos_alta As Double
Dim precio_recam, total_pesos_super_basulip, precio_alta As Double
Dim total_pesos_sup_bas, precio_prom_sup_bas As Double

sql = "SELECT folio,cod_producto,desc_producto,rut_cliente,nom_cliente,peso_neto as kilos,grupo FROM vista_psjes_precios_placilla_2 WHERE fecha_ingreso >= #" & Format(fecha_ini, "yyyy-mm-dd HH:mm") & "# And fecha_ingreso <= #" & Format(fecha_term, "yyyy-mm-dd HH:mm") & "# ORDER BY folio"
total_kilos = 0
precio = 0
row = 1
rst.Open sql, cnt, adOpenForwardOnly, adLockReadOnly
If rst.EOF <> True Then
   grilla.Redraw = False
   Do While Not rst.EOF
      With grilla
          .Rows = row + 1
          .TextMatrix(row, 0) = row
          If Not IsNull(rst!folio) Then .TextMatrix(row, 1) = UCase(rst!folio)
          If Not IsNull(rst!cod_producto) Then .TextMatrix(row, 2) = UCase(rst!cod_producto)
          If Not IsNull(rst!desc_producto) Then .TextMatrix(row, 3) = UCase(rst!desc_producto)
          If Not IsNull(rst!nom_cliente) Then .TextMatrix(row, 4) = UCase(rst!nom_cliente)
          If Not IsNull(rst!kilos) Then .TextMatrix(row, 5) = FormatNumber(UCase(rst!kilos), 0)
          precio = buscar_precio_pl(rst!rut_cliente, rst!cod_producto)
          .TextMatrix(row, 6) = FormatCurrency(precio, 0)
          total_pesos = CDbl(rst!kilos) * precio
          .TextMatrix(row, 7) = FormatCurrency(total_pesos, 0)
          .TextMatrix(row, 8) = rst!grupo
      End With
      Select Case rst!grupo
         Case Is = "RECAM"
             kilos_recam = kilos_recam + rst!kilos
             total_pesos_recam = total_pesos_recam + total_pesos
         Case Is = "ALTA"
             kilos_alta = kilos_alta + rst!kilos
             total_pesos_alta = total_pesos_alta + total_pesos
         Case Is = "SUPER-BASULIP"
             kilos_super_basulip = kilos_super_basulip + rst!kilos

Public Sub LimpiarTablasDePesaje()

        MsgBox "Operación cancelada.", vbExclamation
        Exit Sub
    End If
    
    On Error GoTo ManejoDeErrores

    ' Define las tablas a limpiar

    conectarBD ' Llama a tu función de conexión existente

    ' Bucle que limpia cada tabla
    For Each tabla In tablas
        cnt.Execute "DELETE * FROM " & tabla & ";", , dbFailOnError
    Next tabla

    ' --- CORREGIDO AQUÍ ---
    desconectarBD ' Llama a tu función para desconectar

ManejoDeErrores:

End Sub

Public Sub EjecutarProcedimientosConCommit()
    ' Pide confirmación al usuario
    If MsgBox("¿Deseas ejecutar el proceso completo de inserción (Proveedores y SAP)?", _
              vbYesNo + vbQuestion, "Confirmar Proceso") <> vbYes Then
        MsgBox "Proceso cancelado.", vbExclamation
        Exit Sub
    End If

    On Error GoTo ManejoDeErrores
    
    Application.StatusBar = "Conectando al servidor..."
    DoEvents

    ' Conecta y empieza la transacción
    conectarBdFirebird RUTA_BD_FIREBIRD
    If cntfb Is Nothing Then Exit Sub ' Sale si la conexión falló
    
    cntfb.BeginTrans

    ' Configura y ejecuta los comandos
    Dim cmd As New ADODB.Command
    With cmd
        .ActiveConnection = cntfb
        .CommandType = adCmdStoredProc
        
        ' Ejecuta el primer PA
        Application.StatusBar = "Paso 1 de 2: Ejecutando Inserción de Proveedores..."
        DoEvents
        .CommandText = "PA_INS_PROV_STGO"
        .Execute
        
        ' Ejecuta el segundo PA
        Application.StatusBar = "Paso 2 de 2: Ejecutando Inserción Masiva SAP..."
        DoEvents
        .CommandText = "PA_INS_MASIVO_EXP_SAP"
        .Execute
    End With
    
    ' Confirma los cambios si todo funcionó
    Application.StatusBar = "Confirmando cambios (COMMIT)..."
    DoEvents
    cntfb.CommitTrans

    ' Limpia y notifica
    Application.StatusBar = False
    desconectarBdFirebird
    MsgBox "El proceso completo se ejecutó y confirmó exitosamente.", vbInformation, "Proceso Terminado"
    
    Set cmd = Nothing
    Exit Sub

ManejoDeErrores:
    Application.StatusBar = False
    ' El mensaje de error detallado ya lo da la función de conexión si falla
    ' Este mensaje es para errores durante la ejecución de los PA
    If Not cntfb Is Nothing Then
         MsgBox "Ocurrió un error durante la ejecución:" & vbCrLf & Err.Description & vbCrLf & vbCrLf & _
           "¡IMPORTANTE! No se guardó ningún cambio.", vbCritical, "Operación Fallida"
        
        If cntfb.State = adStateOpen Then cntfb.RollbackTrans
        desconectarBdFirebird
    End If
    Set cmd = Nothing
End Sub

Application.ScreenUpdating = False
Set xlaplica = Excel.Application
Set xllibro = xlaplica.Workbooks.Add(ThisWorkbook.Path & "\Informe_JCRAMM.xls")
Set xlhoja1 = xllibro.Worksheets(1)
xllibro.Worksheets(1).Select

'impresion de resumen por material
row_excel = 14
With xlhoja1
  For row = 1 To grilla2.Rows - 1
    .Cells(row_excel, 3).Value = grilla2.TextMatrix(row, 1)
    .Cells(row_excel, 4).Value = grilla2.TextMatrix(row, 2)
    row_excel = row_excel + 1
    .Rows(row_excel).EntireRow.Insert
  Next row
  .Cells(row_excel, 3) = "Total Kilos"
  .Cells(row_excel, 3).Font.Bold = True
  .Cells(row_excel, 4) = kilos_res
  .Cells(row_excel, 4).Font.Bold = True
  
'impresion de detalle pesajes
row_excel = 8
  For row = 1 To grilla1.Rows - 1
    .Cells(row_excel, 1).Value = Format(grilla1.TextMatrix(row, 1), "yyyy-MM-dd")
    .Cells(row_excel, 2).Value = grilla1.TextMatrix(row, 2)
    .Cells(row_excel, 3).Value = grilla1.TextMatrix(row, 3)
    .Cells(row_excel, 4).Value = grilla1.TextMatrix(row, 4)
    .Cells(row_excel, 5).Value = grilla1.TextMatrix(row, 5)
    .Cells(row_excel, 6).Value = grilla1.TextMatrix(row, 6)
    .Cells(row_excel, 7).Value = grilla1.TextMatrix(row, 7)
    row_excel = row_excel + 1
    .Rows(row_excel).EntireRow.Insert
  Next row
  .Cells(row_excel, 6) = "Total Kilos"
  .Cells(row_excel, 6).Font.Bold = True
  .Cells(row_excel, 7) = kilos_det
  .Cells(row_excel, 7).Font.Bold = True
  .Cells(4, 5) = Format(grilla1.TextMatrix(1, 1), "dd-MMM-yyyy") & " al " & Format(grilla1.TextMatrix(grilla1.Rows - 1, 1), "dd-MMM-yyyy")
End With
xllibro.Worksheets(1).Select
End Sub
